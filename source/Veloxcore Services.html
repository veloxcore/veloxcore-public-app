// Minimal API · .NET 8
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddAzureClients(c =>
{
    c.AddEventHubProducerClient(
        builder.Configuration["EventHub:Conn"],
        "telemetry");
    c.AddBlobServiceClient(
        builder.Configuration["Storage:Conn"]);
});

var app = builder.Build();

app.MapPost("/ingest",
    async (TelemetryEvent ev,
           EventHubProducerClient hub) =>
    {
        await hub.SendAsync(
            new[] { EventData.From(ev) });
        return Results.Accepted();
    })
    .RequireAuthorization()
    .WithName("IngestTelemetry");

app.Run();
